cmake_minimum_required(VERSION 3.22)
project(node_status_host_tests LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(IDF_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.local/esp-idf" CACHE PATH "Pinned ESP-IDF source")
set(CJSON_DIR "${IDF_SOURCE}/components/json/cJSON")
if(NOT EXISTS "${CJSON_DIR}/cJSON.c")
    message(FATAL_ERROR "Install the pinned ESP-IDF source with submodules first; see docs/BUILD.md")
endif()
option(ENABLE_SANITIZERS "Run host tests with AddressSanitizer and UBSan" ON)
if(ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

add_library(cjson STATIC "${CJSON_DIR}/cJSON.c")
target_include_directories(cjson PUBLIC "${CJSON_DIR}")
add_library(node_status STATIC ../components/node_status/node_status.c)
target_include_directories(node_status PUBLIC ../components/node_status/include)
target_link_libraries(node_status PUBLIC cjson)
target_compile_options(node_status PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -pedantic)
add_executable(test_node_status test_node_status.c)
target_link_libraries(test_node_status PRIVATE node_status)
target_compile_options(test_node_status PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -pedantic)

add_library(wifi_scan_logic STATIC ../components/wifi_service/wifi_scan_logic.c)
target_include_directories(wifi_scan_logic PUBLIC ../components/wifi_service/include)
target_link_libraries(wifi_scan_logic PUBLIC cjson)
target_compile_options(wifi_scan_logic PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -pedantic)
add_executable(test_wifi_scan test_wifi_scan.c)
target_link_libraries(test_wifi_scan PRIVATE wifi_scan_logic)
target_compile_options(test_wifi_scan PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -pedantic)

add_library(wifi_link_logic STATIC ../components/wifi_service/wifi_link_logic.c)
target_include_directories(wifi_link_logic PUBLIC ../components/wifi_service/include)

target_compile_options(wifi_link_logic PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -pedantic)
add_executable(test_wifi_link test_wifi_link.c)
target_link_libraries(test_wifi_link PRIVATE wifi_link_logic)

add_library(wifi_provision STATIC ../components/wifi_service/wifi_provision.c)

target_compile_options(wifi_provision PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -pedantic)
add_executable(test_wifi_provision test_wifi_provision.c)
target_link_libraries(test_wifi_provision PRIVATE wifi_provision)
target_compile_options(test_wifi_provision PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -pedantic)

add_executable(test_wifi_credentials test_wifi_credentials.c)
target_link_libraries(test_wifi_credentials PRIVATE wifi_provision)
target_compile_options(test_wifi_credentials PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -pedantic)

enable_testing()
add_library(ota_policy STATIC ../components/ota_service/ota_policy.c)
target_include_directories(ota_policy PUBLIC ../components/ota_service/include)
target_compile_options(ota_policy PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -pedantic)
add_executable(test_ota_policy test_ota_policy.c)
target_link_libraries(test_ota_policy PRIVATE ota_policy)
add_test(NAME ota_policy COMMAND test_ota_policy)
foreach(case valid malformed bounds)
    add_test(NAME wifi_credentials_${case} COMMAND test_wifi_credentials ${case})
endforeach()
foreach(case initial lifecycle backoff flapping invalid limits json bounds allocation)
    add_test(NAME wifi_link_${case} COMMAND test_wifi_link ${case})
endforeach()
foreach(case valid malformed controls)
    add_test(NAME wifi_provision_${case} COMMAND test_wifi_provision ${case})
endforeach()
foreach(case initial transitions invalid records octets bounds allocation)
    add_test(NAME wifi_scan_${case} COMMAND test_wifi_scan ${case})
endforeach()
foreach(case initial transitions ordering escaping bounds invalid extremes allocation)
    add_test(NAME status_${case} COMMAND test_node_status ${case})
endforeach()
