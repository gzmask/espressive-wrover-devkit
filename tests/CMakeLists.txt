option(ENABLE_LISPBM_TESTS "Build real pinned LispBM integration tests" OFF)
if(ENABLE_LISPBM_TESTS)
    include("${CMAKE_CURRENT_LIST_DIR}/../cmake/lispbm.cmake")
    wrover_lispbm_sources(lbm_sources)
    add_library(lispbm STATIC ${lbm_sources})
    target_include_directories(lispbm PUBLIC "${LISPBM_SOURCE_DIR}/include" ../components/lispbm_runtime/port)
    target_compile_definitions(lispbm PUBLIC LBM_SINGLE_THREADED)
    option(LISPBM_HOST_WORD32 "Test 32-bit LispBM values in low-address memory on a 64-bit host (not Xtensa)" OFF)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT LISPBM_HOST_WORD32)
        target_compile_definitions(lispbm PUBLIC LBM64)
    elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
        target_compile_definitions(lispbm PUBLIC TEST_LBM32_LOW_MEMORY)
        if(APPLE)
            if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64")
                message(FATAL_ERROR "macOS arm64 cannot launch a reduced-PAGEZERO binary; use -DCMAKE_OSX_ARCHITECTURES=x86_64 with existing Rosetta for the word32 model")
            endif()
            # Test executables only: allow a low mmap hint without MAP_FIXED.
            add_link_options(-Wl,-pagezero_size,0x1000)
        else()
            target_compile_definitions(lispbm PUBLIC _DEFAULT_SOURCE)
        endif()
    endif()
    add_library(lisp_runtime STATIC ../components/lisp_repl/lisp_runtime.c)
    target_include_directories(lisp_runtime PUBLIC ../components/lisp_repl/include)
    target_link_libraries(lisp_runtime PUBLIC lispbm lcd_terminal)
    target_compile_options(lisp_runtime PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
    add_executable(test_lisp_runtime test_lisp_runtime.c)
    target_link_libraries(test_lisp_runtime PRIVATE lisp_runtime)
    add_executable(test_repl_session test_repl_session.c)
    target_link_libraries(test_repl_session PRIVATE lisp_runtime repl_frontend)
    add_library(lisp_runtime_faults STATIC ../components/lisp_repl/lisp_runtime.c)
    target_include_directories(lisp_runtime_faults PUBLIC ../components/lisp_repl/include)
    target_link_libraries(lisp_runtime_faults PUBLIC lispbm lcd_terminal)
    foreach(name memory_init symrepr_init heap_init print_init extensions_init eval_init image_boot add_extension)
        target_compile_definitions(lisp_runtime_faults PRIVATE lbm_${name}=fault_lbm_${name})
    endforeach()
    target_compile_options(lisp_runtime_faults PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
    add_executable(test_lisp_init test_lisp_init.c)
    target_link_libraries(test_lisp_init PRIVATE lisp_runtime_faults)
endif()

    endforeach()

        # visible, but do not promote them to errors in these test targets.
        foreach(target lisp_runtime lisp_runtime_faults test_lisp_init)
            target_compile_options(${target} PRIVATE -Wno-error=pointer-to-int-cast -Wno-error=int-to-pointer-cast)
        endforeach()
    endif()
endif()
